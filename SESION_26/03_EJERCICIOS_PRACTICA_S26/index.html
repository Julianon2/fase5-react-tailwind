import { Link, useLocation } from "react-router-dom";
import { useContext } from "react";
import { CartContext } from "./CartContext.jsx";
import { AuthContext } from "./AuthContext.jsx";

function Navbar() {
  const ubicacion = useLocation();
  const { items } = useContext(CartContext);
  const { usuario } = useContext(AuthContext);
  return (
    <nav className="flex items-center justify-between px-8 py-4 bg-white border-b border-slate-200">
      <div className="text-xl font-extrabold text-verde">TechStore Pro</div>
      <ul className="hidden md:flex gap-6 text-sm font-semibold text-texto-dim">
        <li>
          <Link to="/" className={ubicacion.pathname === "/" ? "text-verde font-extrabold" : ""}>
            Inicio
          </Link>
        </li>
        <li>
          <Link to="/productos" className={ubicacion.pathname === "/productos" ? "text-verde font-extrabold" : ""}>
            Productos
          </Link>
        </li>
        <li>Nosotros</li>
        <li>Contacto</li>
      </ul>
      <div className="flex items-center gap-4">
        <span className="text-sm font-semibold text-texto-dim">
          Carrito: {items.length}
        </span>
        {usuario ? (
          <span className="text-sm font-bold text-verde">

          </span>

            Ingresar
          </button>
        )}
      </div>
    </nav>
  );
}

export default Navbar;
ParteQué hace
useContext(AuthContext)Lee el valor transmitido por AuthProvider — funciona porque main.jsx ya envuelve la app (guía 02, Paso 3)
{ usuario }El nombre guardado, o null si nadie ha iniciado sesión
{usuario ? (...) : (...)}Operador ternario — si hay usuario, muestra "Hola, nombre"; si no, muestra el botón "Ingresar" de siempre

Ejercicio 3

BadgeCarrito.jsx — un componente propio

Hasta ahora, useContext siempre se usó dentro de Navbar.jsx. Este ejercicio separa esa lógica en su propio componente — practicando que cualquier componente, en cualquier archivo, puede leer el mismo Context.

1

Crear src/BadgeCarrito.jsx con este contenido completo:

frontend-react/src/BadgeCarrito.jsxcontenido completo
import { useContext } from "react";
import { CartContext } from "./CartContext.jsx";

function BadgeCarrito() {
  const { items } = useContext(CartContext);

  if (items.length === 0) return null;

  return (
    <span className="bg-verde text-white text-xs font-bold rounded-full px-2 py-0.5">
      {items.length}
    </span>
  );
}

export default BadgeCarrito;
ParteQué hace
useContext(CartContext) en un archivo nuevoConfirma que el Context funciona en CUALQUIER componente, no solo en Navbar
if (items.length === 0) return nullSi el carrito está vacío, el componente no dibuja nada — ni siquiera un span vacío
2
