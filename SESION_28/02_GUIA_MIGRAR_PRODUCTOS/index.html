Paso 1 · Componente

Crear ProductoModal.jsx

El modal es un componente más: recibe el producto que debe mostrar y una función para avisar que quiere cerrarse. Se escribe en un archivo propio, igual que ProductCard. Todavía no se conecta a nada: primero la ficha, después la conexión.

1

Levantar los 2 servidores, cada uno en su propia terminal (si ya estaban corriendo, no hace falta reiniciarlos):

terminal 1 — dentro de backend/
npm run dev
terminal 2 — dentro de frontend-react/
npm run dev
Cómo verificar: localhost:5173/productos muestra las tarjetas con foto a sangre, el buscador y el botón "Agregar +", como quedó en S27.
2

Crear src/ProductoModal.jsx con este contenido completo:

frontend-react/src/ProductoModal.jsxarchivo nuevo — contenido completo
import { useContext } from "react";
import { CartContext } from "./CartContext.jsx";

function ProductoModal({ producto, onCerrar }) {
  const { agregarAlCarrito } = useContext(CartContext);

  function agregar() {
    agregarAlCarrito({
      id: producto.id,
      nombre: producto.nombre,
      precio: producto.precio,
      imagen: producto.imagen,
    });
    onCerrar();
  }

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-ink/60 p-4">
      <div
        role="dialog"
        aria-modal="true"
        aria-label={producto.nombre}
        className="relative w-full max-w-3xl max-h-[90vh] overflow-y-auto rounded-3xl bg-paper shadow-float md:grid md:grid-cols-2"
      >
        <button
          onClick={onCerrar}
          aria-label="Cerrar"
          className="absolute top-4 right-4 z-10 size-10 rounded-full bg-paper/90 text-xl leading-none flex items-center justify-center hover:bg-accent hover:text-paper transition-colors"
        >
          ×
        </button>

        <img
          src={producto.imagen.replace("w=400&h=250", "w=800&h=1000")}
          alt={producto.nombre}
          className="w-full h-64 md:h-full object-cover"
        />

        <div className="flex flex-col gap-4 p-6 md:p-8">
          <span className="font-mono text-[11px] font-bold tracking-wider text-gray-500">
            REF {String(producto.id).padStart(2, "0")}
          </span>
          <h2 className="font-display font-bold text-3xl tracking-tight leading-tight">
            {producto.nombre}
          </h2>
          <p className="text-gray-700 leading-relaxed">{producto.descripcion}</p>
          <p className="font-mono text-2xl font-bold">{producto.precio}</p>
          <button
            onClick={agregar}
            className="mt-auto bg-ink text-paper rounded-card py-3.5 text-sm font-semibold hover:bg-accent transition-colors"
          >
            Agregar al carrito
          </button>
        </div>
      </div>
    </div>
  );
}

export default ProductoModal;

frontend-react/src/Productos.jsxantes
import { useState, useEffect } from "react";
import ProductCard from "./ProductCard.jsx";

function Productos() {
  const [busqueda, setBusqueda] = useState("");
  const [productos, setProductos] = useState([]);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState(null);

  function cargarProductos() {
    setCargando(true);
    setError(null);
    fetch("http://localhost:3000/api/productos")
      .then((res) => res.json())
      .then((datos) => setProductos(datos))
      .catch(() => setError("No se pudo conectar al servidor"))
      .finally(() => setCargando(false));
  }

  useEffect(() => {
    cargarProductos();
  }, []);

  const filtrados = productos.filter((p) =>
    p.nombre.toLowerCase().includes(busqueda.toLowerCase()),
  );

  return (
    <div className="flex flex-col gap-10">
      <div className="flex flex-wrap items-center gap-3">
        <input
          type="text"
          placeholder="Buscar producto..."
          value={busqueda}
          onChange={(e) => setBusqueda(e.target.value)}
          className="w-full max-w-md px-4 py-2 border border-gray-300 rounded-card focus:outline-none focus:border-ink"
        />
        {busqueda && (
          <p className="text-sm text-gray-500 whitespace-nowrap">
            {filtrados.length} {filtrados.length === 1 ? "resultado" : "resultados"}
          </p>
        )}
      </div>
      {cargando && <p className="text-center py-10">Cargando productos...</p>}
      {error && <p className="text-center text-accent-dark py-10">{error}</p>}
      {!cargando && !error && (
        <section className="grid grid-cols-1 sm:grid md:grid-cols-2 lg:grid-cols-3 gap-6">
          {filtrados.map((p) => (
            <ProductCard
              key={p.id}
              nombre={p.nombre}
              descripcion={p.descripcion}
              precio={p.precio}
              imagen={p.imagen}
              id={p.id}
            />
          ))}
        </section>
      )}
      {!cargando && !error && filtrados.length === 0 && (
        <p className="text-center text-gray-500 py-10">
          No se encontraron productos con ese nombre.
        </p>
      )}
    </div>
  );
}

export default Productos;

Paso 3 · Eventos

function precioNumero(p) {
  return Number(p.precio.replace(/\D/g, ""));
}

function Productos() {
  const [busqueda, setBusqueda] = useState("");
  const [productos, setProductos] = useState([]);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState(null);
  const [seleccionado, setSeleccionado] = useState(null);

      .then((res) => res.json())
