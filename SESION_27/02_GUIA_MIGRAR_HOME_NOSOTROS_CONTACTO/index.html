import { useState, useContext } from "react";
import { Link, NavLink } from "react-router-dom";
import { CartContext } from "./CartContext.jsx";
import { AuthContext } from "./AuthContext.jsx";

const enlaces = [
  { ruta: "/", texto: "Inicio" },
  { ruta: "/productos", texto: "Productos" },
  { ruta: "/nosotros", texto: "Nosotros" },
  { ruta: "/contacto", texto: "Contacto" },
];

function Navbar() {
  const { items, vaciarCarrito } = useContext(CartContext);
  const { usuario } = useContext(AuthContext);
  const [abierto, setAbierto] = useState(false);

  function cerrar() {
    setAbierto(false);
  }

  return (
    <nav className="w-full max-w-6xl mx-auto px-6 pt-6">
      <div className="flex items-center justify-between gap-4">
        <Link to="/" onClick={cerrar} className="font-wordmark text-lg md:text-xl tracking-wide">
          TECH<span className="text-accent-dark">STORE</span>
        </Link>

        <ul className="hidden md:flex gap-9 text-sm font-medium text-gray-700">
          {enlaces.map((enlace) => (
            <li key={enlace.ruta}>
              <NavLink
                to={enlace.ruta}
                end
                className={({ isActive }) =>
                  isActive ? "text-ink font-semibold" : "hover:text-ink transition-colors"
                }
              >
                {enlace.texto}
              </NavLink>
            </li>
          ))}
        </ul>

        <div className="flex items-center gap-4">
          <span className="font-mono text-xs md:text-[13px] text-gray-700 whitespace-nowrap">
            Carrito ( {items.length} )
          </span>
          {items.length > 0 && (
            <button
              onClick={vaciarCarrito}
              className="hidden md:block text-xs text-accent-dark underline"
            >
              Vaciar
            </button>
          )}
          <button className="hidden md:block bg-ink text-paper px-5 py-2.5 rounded-card text-sm font-semibold hover:bg-accent transition-colors">
            {usuario ? usuario : "Ingresar"}
          </button>

          <button
            onClick={() => setAbierto(!abierto)}
            aria-label={abierto ? "Cerrar menú" : "Abrir menú"}
            aria-expanded={abierto}
            className="md:hidden relative size-10 -mr-2"
          >
            <span
              className={
                "absolute left-2 top-1/2 h-0.5 w-6 -mt-px rounded-full bg-ink transition-all duration-300 " +

              }
            ></span>
            <span
              className={
                "absolute left-2 top-1/2 h-0.5 w-6 -mt-px rounded-full bg-ink transition-all duration-300 " +

              }
            ></span>
            <span
              className={
                "absolute left-2 top-1/2 h-0.5 w-6 -mt-px rounded-full bg-ink transition-all duration-300 " +

              }
            ></span>
          </button>
        </div>
      </div>

      <div
        inert={!abierto}
        className={
          "md:hidden grid transition-[grid-template-rows] duration-300 ease-out " +
          (abierto ? "grid-rows-[1fr]" : "grid-rows-[0fr]")
        }
      >
        <div className="overflow-hidden">
          <ul className="flex flex-col pt-6 pb-4">
            {enlaces.map((enlace, i) => (
              <li
                key={enlace.ruta}
                style={{ transitionDelay: abierto ? i * 60 + 100 + "ms" : "0ms" }}
                className={
                  "transition-all duration-300 " +

                {usuario ? usuario : "Ingresar"}
              </button>
            </li>
          </ul>
        </div>
      </div>
    </nav>
  );

import { useState, useContext }Se suma useState (S23) para recordar si el menú de celular está abierto o cerrado
import { Link, NavLink }NavLink es un Link que sabe si apunta a la página actual. Reemplaza al useLocation + ternario repetido en cada enlace
const enlaces = [ ... ]Un array con los 4 enlaces, definido fuera del componente porque nunca cambia. Se usa dos veces (menú de escritorio y menú de celular): agregar un enlace nuevo es agregar una línea

function cerrar() { setAbierto(false) }Cierra el menú. Se llama al elegir un enlace, para que el menú no quede abierto sobre la página nueva

font-wordmark text-lg md:text-xl tracking-wideMichroma. Un tamaño más chico en celular (text-lg, 18px) para que quepa junto al carrito y la hamburguesa, y 20px desde md. No uses font-bold: Michroma solo tiene un grosor y el navegador fingiría la negrita deformando las letras
TECH<span className="text-accent-dark">STORE</span>Solo "STORE" va en coral. Se usa accent-dark (más oscuro que accent) porque con un trazo tan fino el coral claro pierde contraste sobre blanco. Sin logo ni ícono junto a la marca

className={({ isActive }) => ...}NavLink permite que className sea una función que recibe isActive. Si es la página actual, negro y grueso; si no, gris con hover
font-mono text-xs md:text-[13px] ... whitespace-nowrapJetBrains Mono para el contador. whitespace-nowrap evita que "Carrito ( 0 )" se parta en dos líneas en pantallas angostas
hidden md:block (Vaciar e Ingresar)En celular esos dos botones no caben en la fila, así que se ocultan y "Ingresar" pasa al panel del menú. Desde md se ven en la fila
<button onClick={() => setAbierto(!abierto)} ...>El botón de la hamburguesa. md:hidden lo muestra solo en celular. Al hacer clic invierte el estado: abierto pasa a cerrado y al revés
aria-label y aria-expandedLe dicen a los lectores de pantalla qué hace el botón ("Abrir menú" o "Cerrar menú") y si el menú está desplegado
los 3 <span> (las rayas)Cada raya es un <span> de 2px de alto y 24px de ancho, colocado en el centro con absolute top-1/2. Se dibujan con CSS: no hace falta ninguna librería de íconos
-translate-y-2 / translate-y-2En reposo la raya de arriba sube 8px y la de abajo baja 8px, formando las 3 rayas

transition-all duration-300Anima cualquier cambio de esas clases durante 300 milisegundos. Es lo que convierte el salto en un movimiento suave

inert={!abierto}Mientras el menú está cerrado, sus enlaces no se pueden enfocar con la tecla Tab ni leer con lectores de pantalla, aunque estén ocultos

text-3xl font-display ... text-gray-500Enlaces grandes en Space Grotesk. El de la página actual va en negro y los demás en gris

const express = require('express');

// 2. POST /api/contacto — público: cualquier visitante puede escribir
router.post('/', async (req, res) => {
  try {
    // Solo se toman los 5 campos esperados; cualquier otro campo se ignora
    const { nombre, email, telefono, asunto, mensaje } = req.body;

    await Mensaje.create({ nombre, email, telefono, asunto, mensaje });

    res.status(201).json({ mensaje: 'Mensaje recibido' });
  } catch (err) {
    // Si falló una regla del modelo, se devuelve el primer error en español
    if (err.name === 'ValidationError') {
      const primero = Object.values(err.errors)[0].message;
      return res.status(400).json({ error: primero });
    }
    res.status(500).json({ error: 'Error al guardar el mensaje' });
  }
});

// 3. Exportar el router
module.exports = router;
