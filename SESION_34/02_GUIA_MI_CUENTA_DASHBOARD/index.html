Paso 2 · El hook

Sacar el efecto repetido a una función propia

En la S33, MisPedidos pide sus datos con un efecto largo: el token, el 401 y el 403, los tres estados. Hoy esa misma lógica la necesitan tres páginas. En vez de copiarla, se extrae a un hook propio, useApi, que cualquier página puede usar con una sola línea.

1

Crea el archivo frontend-react/src/hooks/useApi.js con este contenido completo:

import { useState, useEffect, useContext } from "react";
import { useNavigate, useLocation } from "react-router-dom";
import { AuthContext } from "../context/AuthContext.jsx";

// Pide datos al backend con el token de la sesión.
// Devuelve { datos, cargando, error } para que cada página decida qué dibujar.
function useApi(ruta) {
  const { logout } = useContext(AuthContext);
  const navegar = useNavigate();
  const ubicacion = useLocation();
  const [datos, setDatos] = useState(null);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    async function cargar() {
      try {
        const respuesta = await fetch("http://localhost:3000" + ruta, {
          headers: { Authorization: "Bearer " + localStorage.getItem("token") },
        });

        // 401: no hay token. 403: el token ya no sirve (venció o está mal).
        // Se cierra la sesión y se manda a ingresar, para volver después a esta misma página.
        if (respuesta.status === 401 || respuesta.status === 403) {
          logout();
          navegar("/login", {
            state: {
              aviso: "Tu sesión terminó. Ingresa de nuevo para continuar",
              desde: ubicacion.pathname,
            },
          });
          return;
        }
        if (!respuesta.ok) throw new Error("respuesta no válida");

        setDatos(await respuesta.json());
      } catch {
        setError("No se pudieron cargar los datos. Revisa que el backend esté encendido.");
      } finally {
        setCargando(false);
      }
    }
    cargar();
  }, [ruta]);

  return { datos, cargando, error };
}

export default useApi;

// Barras horizontales hechas solo con CSS. Cada barra mide su valor frente al mayor de todos.
function GraficoBarras({ titulo, datos }) {
  const maximo = Math.max(...datos.map((dato) => dato.valor), 1);

  return (
    <div className="border border-gray-300 rounded-card p-6 flex flex-col gap-5">
      <h3 className="font-display font-bold text-lg">{titulo}</h3>
      <ul className="flex flex-col gap-3">
        {datos.map((dato) => (
          <li key={dato.etiqueta} className="flex items-center gap-3 text-sm">
            <span className="w-32 shrink-0 text-gray-700">{dato.etiqueta}</span>
            <span className="flex-1 h-3 rounded-full bg-gray-100">
              <span
                className="block h-3 rounded-full bg-accent"
                style={{ width: (dato.valor / maximo) * 100 + "%" }}
              ></span>
            </span>
            <span className="w-6 text-right font-mono">{dato.valor}</span>
          </li>
        ))}
      </ul>
    </div>
  );
}

  PENDING: { titulo: "Confirmando tu pago", texto: "Estamos esperando la respuesta de Wompi. No cierres esta página." },
  APPROVED: { titulo: "Pago aprobado", texto: "Tu compra quedó registrada y tu carrito está vacío." },
  DECLINED: { titulo: "Pago rechazado", texto: "El banco no aprobó el pago. Tu carrito sigue como estaba: puedes intentar con otro medio de pago." },
  ERROR: { titulo: "El pago no se pudo completar", texto: "Hubo un error en el proceso. Tu carrito sigue como estaba." },
  VOIDED: { titulo: "El pago fue anulado", texto: "La transacción se anuló. Tu carrito sigue como estaba." },
  TIMEOUT: { titulo: "Todavía no tenemos respuesta", texto: "No pudimos confirmar el pago en un minuto. Si cerraste la ventana sin pagar, tu carrito sigue como estaba." },
};

function PagoResultado() {
  const { referencia } = useParams();
  const { vaciarCarrito } = useContext(CartContext);
  const [estado, setEstado] = useState("PENDING");

  // Pregunta al servidor cada 3 segundos, hasta 20 veces (un minuto)
  useEffect(() => {
    let intentos = 0;
    const intervalo = setInterval(async () => {
      intentos++;
      if (intentos > 20) {
        clearInterval(intervalo);
        setEstado("TIMEOUT");
        return;
      }
      try {
        const respuesta = await fetch("http://localhost:3000/api/pagos/estado/" + referencia, {
          headers: { Authorization: "Bearer " + localStorage.getItem("token") },
        });
        const datos = await respuesta.json();
        if (datos.status && datos.status !== "PENDING") {
          clearInterval(intervalo);
          setEstado(datos.status);
        }
      } catch (error) {
        console.error("No se pudo consultar el pago", error);
      }
    }, 3000);

    // Limpieza: si la persona se va de la página, el reloj se detiene
    return () => clearInterval(intervalo);
  }, [referencia]);

  // Cuando el pago se aprueba, el carrito se vacía
  useEffect(() => {
    if (estado === "APPROVED") vaciarCarrito();
  }, [estado]);

  const mensaje = MENSAJES[estado] || MENSAJES.ERROR;

  return (
    <section className="max-w-md mx-auto text-center flex flex-col items-center gap-5 py-16">
      <p className="font-mono text-xs uppercase tracking-widest text-gray-500">Pago</p>
      <h1 className="font-display font-bold text-4xl tracking-tighter">
        {mensaje.titulo}
        <span className="text-accent">.</span>
      </h1>
      <p className="text-gray-700">{mensaje.texto}</p>
      <p className="font-mono text-xs text-gray-500">Referencia: {referencia}</p>

  VOIDED: { titulo: "El pago fue anulado", texto: "La transacción se anuló. Tu carrito sigue como estaba." },
  TIMEOUT: { titulo: "Todavía no tenemos respuesta", texto: "No pudimos confirmar el pago en un minuto. Si cerraste la ventana sin pagar, tu carrito sigue como estaba." },
};

function PagoResultado() {
  const { referencia } = useParams();
  const { vaciarCarrito } = useContext(CartContext);
  const [estado, setEstado] = useState("PENDING");

  // Pregunta al servidor cada 3 segundos, hasta 20 veces (un minuto)
  useEffect(() => {
    let intentos = 0;
    const intervalo = setInterval(async () => {
      intentos++;
      if (intentos > 20) {
        clearInterval(intervalo);
        setEstado("TIMEOUT");
        return;
      }
      try {
        const respuesta = await fetch("http://localhost:3000/api/pagos/estado/" + referencia, {
          headers: { Authorization: "Bearer " + localStorage.getItem("token") },
        });
        const datos = await respuesta.json();
        if (datos.status && datos.status !== "PENDING") {
          clearInterval(intervalo);
          setEstado(datos.status);
        }
      } catch (error) {
        console.error("No se pudo consultar el pago", error);
      }
    }, 3000);

    // Limpieza: si la persona se va de la página, el reloj se detiene
    return () => clearInterval(intervalo);
  }, [referencia]);

  // Cuando el pago se aprueba, el carrito se vacía
  useEffect(() => {
    if (estado === "APPROVED") vaciarCarrito();
  }, [estado]);

  const mensaje = MENSAJES[estado] || MENSAJES.ERROR;

  return (
    <section className="max-w-md mx-auto text-center flex flex-col items-center gap-5 py-16">
      <p className="font-mono text-xs uppercase tracking-widest text-gray-500">Pago</p>
      <h1 className="font-display font-bold text-4xl tracking-tighter">
        {mensaje.titulo}
        <span className="text-accent">.</span>
      </h1>
      <p className="text-gray-700">{mensaje.texto}</p>
      <p className="font-mono text-xs text-gray-500">Referencia: {referencia}</p>

      {estado === "APPROVED" && (
        <div className="flex flex-wrap justify-center gap-3">
          <Link
            to="/cuenta/pedidos"
            className="bg-ink text-paper px-6 py-3 rounded-card text-sm font-semibold hover:bg-accent transition-colors"
          >
            Ver mis pedidos
          </Link>
          <Link
            to="/productos"
            className="px-6 py-3 rounded-card text-sm font-semibold border border-gray-300 hover:border-ink transition-colors"
          >
            Seguir comprando
          </Link>
        </div>
      )}
      {estado !== "PENDING" && estado !== "APPROVED" && (
        <Link
          to="/carrito"
          className="bg-ink text-paper px-6 py-3 rounded-card text-sm font-semibold hover:bg-accent transition-colors"
        >
          Volver al carrito
        </Link>
      )}
    </section>
  );
}

export default PagoResultado;
