Solo correos de proveedores conocidos

El registro de TechStore Pro hoy acepta cualquier correo con forma de correo. En esta guía aprendes a leer el dominio (lo que va después de la arroba) y a comparar con una lista de permitidos, en el servidor y en el formulario. Es un complemento corto de la guía 02.

Paso 1 · Backend

El servidor revisa el dominio

Hoy el registro acepta cualquier correo con forma de correo, como ana@correo.com o x@x.xx. Queremos aceptar solo los de proveedores conocidos. El dominio es lo que va después de la arroba: en ana@gmail.com el dominio es gmail.com. La regla se pone primero en el servidor, que es el que nadie puede saltarse. Se cambia un solo archivo: backend/routes/auth.js.

1

Este es tu backend/routes/auth.js real, completo, tal como está hoy:

// 1. Importar dependencias
const express  = require('express');
const bcrypt   = require('bcryptjs');
const jwt      = require('jsonwebtoken');
const Usuario  = require('../models/Usuario');
const router   = express.Router();

// 2. POST /api/auth/registro — crear cuenta nueva
router.post('/registro', async (req, res) => {
  try {
    // El rol NO se lee del cuerpo: lo que manda el navegador no es de fiar
    const { nombre, email, password, departamento, municipio } = req.body;

    // Validar los datos: el servidor revisa aunque la página ya lo haya hecho
    if (!nombre || nombre.trim().length < 3) {
      return res.status(400).json({ error: 'El nombre debe tener al menos 3 caracteres' });
    }
    if (!email || !/^\S+@\S+\.\S+$/.test(email)) {
      return res.status(400).json({ error: 'El correo no es válido' });
    }
    if (!password || password.length < 6) {
      return res.status(400).json({ error: 'La contraseña debe tener al menos 6 caracteres' });
    }

    // Verificar que el email no exista ya
    const existe = await Usuario.findOne({ email });
    if (existe) return res.status(400).json({ error: 'El email ya está registrado' });

    // Encriptar la contraseña con 10 rondas de bcrypt
    const hash = await bcrypt.hash(password, 10);

    // Guardar el usuario con la contraseña encriptada
    const usuario = await Usuario.create({ nombre, email, password: hash, departamento, municipio, rol: 'cliente' });

    res.status(201).json({ mensaje: 'Usuario creado correctamente', id: usuario._id });
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// 3. POST /api/auth/login — iniciar sesión y recibir token
router.post('/login', async (req, res) => {
  try {
    const { email, password } = req.body;

    // Buscar usuario por email
    const usuario = await Usuario.findOne({ email });
    if (!usuario) return res.status(401).json({ error: 'Email o contraseña incorrectos' });

    // Comparar la contraseña con el hash guardado en Atlas
    const valida = await bcrypt.compare(password, usuario.password);
    if (!valida) return res.status(401).json({ error: 'Email o contraseña incorrectos' });

    // Crear el token JWT — dura 24 horas
    const token = jwt.sign(
      { id: usuario._id, email: usuario.email, rol: usuario.rol },
      process.env.JWT_SECRET,
      { expiresIn: '24h' }
    );

    res.json({ token, nombre: usuario.nombre, rol: usuario.rol });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// 4. GET /api/auth/perfil — devuelve los datos del usuario logueado (requiere token)
const verificarToken = require('../middleware/auth');

router.get('/perfil', verificarToken, async (req, res) => {
  try {
    // req.usuario.id viene del token — lo usa verificarToken
    // select('-password') excluye el hash de contraseña de la respuesta
    const usuario = await Usuario.findById(req.usuario.id).select('-password');
    if (!usuario) return res.status(404).json({ error: 'Usuario no encontrado' });
    res.json(usuario);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// 5. Exportar el router
module.exports = router;

ParteQué hace
const dominiosPermitidos = [ ... ]Una lista (arreglo) con los dominios que aceptamos. Está fuera del registro para poder verla y cambiarla en un solo lugar
email.lastIndexOf('@')Busca la posición de la última arroba del correo. En ana@gmail.com es la posición 3
email.slice(posición + 1)Corta el texto desde ese punto hasta el final: lo que queda es el dominio, gmail.com. El + 1 salta la arroba
.toLowerCase()Pasa el dominio a minúsculas. Así ANA@GMAIL.COM también se acepta, porque los dominios no distinguen mayúsculas
dominiosPermitidos.includes(dominio)Responde true si el dominio está en la lista. El ! de la condición lo invierte: entra al if cuando no está
return res.status(400).json({ error: '...' })Si el dominio no está permitido, el servidor responde 400 con un mensaje en español y deja de procesar la petición
¿Por qué la última arroba? Si alguien manda ana@gmail.com@evil.com, con lastIndexOf el dominio es evil.com y se rechaza. Si se tomara la primera arroba, el dominio sería gmail.com y pasaría. Elegir bien dónde cortar es parte de validar.
Fíjate en el orden: primero se revisa que el correo tenga forma de correo y después su dominio. El nodemon reinicia el backend solo al guardar.

El servidor ya protege, pero el aprendiz tendría que enviar el formulario para enterarse. Se agrega la misma regla en Registro.jsx, dentro de validar(), para que el error salga en rojo bajo el campo, sin viajar al servidor. Se cambia un solo archivo.

1

Este es tu Registro.jsx real, completo, tal como está hoy:

frontend-react/src/Registro.jsxantes
import { useState, useEffect } from "react";
import { Link, useNavigate } from "react-router-dom";
import Campo, { claseInput } from "./Campo.jsx";

const URL_API = "http://localhost:3000/api/auth";
const URL_COLOMBIA = "https://api-colombia.com/api/v1";

const FORM_VACIO = {
  nombre: "",
  email: "",
  password: "",
  confirmar: "",
  departamento: "",
  municipio: "",
};

function porNombre(a, b) {
  return a.name.localeCompare(b.name);
}

function Registro() {
  const navegar = useNavigate();
  const [form, setForm] = useState(FORM_VACIO);
  const [errores, setErrores] = useState({});
  const [estado, setEstado] = useState("inactivo");
  const [mensajeError, setMensajeError] = useState("");
  const [departamentos, setDepartamentos] = useState([]);
  const [municipios, setMunicipios] = useState([]);

  const elegido = departamentos.find((d) => d.name === form.departamento);
  const departamentoId = elegido ? elegido.id : null;

  useEffect(() => {
    fetch(URL_COLOMBIA + "/Department")
      .then((res) => res.json())
      .then((datos) => setDepartamentos(datos.sort(porNombre)))
      .catch(() => setDepartamentos([]));
  }, []);

  useEffect(() => {
    if (!departamentoId) {
      setMunicipios([]);
      return;
    }
    let vigente = true;
    fetch(URL_COLOMBIA + "/Department/" + departamentoId + "/cities")
      .then((res) => res.json())
      .then((datos) => {
        if (vigente) setMunicipios(datos.sort(porNombre));
      })
      .catch(() => {
        if (vigente) setMunicipios([]);
      });
    return () => {
      vigente = false;
    };
  }, [departamentoId]);

  function manejarCambio(e) {
    const { name, value } = e.target;
    const nuevo = { ...form, [name]: value };
    if (name === "departamento") nuevo.municipio = "";
    setForm(nuevo);
    setErrores({ ...errores, [name]: "" });
    if (estado !== "inactivo") setEstado("inactivo");
  }

  function validar() {
    const nuevos = {};
    if (form.nombre.trim().length < 3) {
      nuevos.nombre = "Escribe tu nombre completo (mínimo 3 caracteres)";
    }
    if (!/^\S+@\S+\.\S+$/.test(form.email)) {
      nuevos.email = "Escribe un correo válido, como nombre@correo.com";
    }
    if (form.password.length < 6) {
      nuevos.password = "La contraseña debe tener al menos 6 caracteres";
    }
    if (form.confirmar !== form.password) {
      nuevos.confirmar = "Las contraseñas no coinciden";
    }
    if (!form.departamento) {
      nuevos.departamento = "Selecciona un departamento";
    }
    if (!form.municipio) {
      nuevos.municipio = "Selecciona un municipio";
    }
    return nuevos;
  }

  async function manejarEnvio(e) {
    e.preventDefault();

    const nuevos = validar();
    setErrores(nuevos);
    if (Object.keys(nuevos).length > 0) return;

    setEstado("enviando");
    try {
      const res = await fetch(URL_API + "/registro", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          nombre: form.nombre,
          email: form.email,
          password: form.password,
          departamento: form.departamento,
          municipio: form.municipio,
        }),
      });
      const datos = await res.json();
      if (!res.ok) throw new Error(datos.error || "No se pudo crear la cuenta");

      navegar("/login", {
        state: { aviso: "Cuenta creada. Ya puedes iniciar sesión." },
      });
    } catch (err) {
      setMensajeError(
        err instanceof TypeError ? "No se pudo conectar al servidor" : err.message,
      );
      setEstado("error");
    }
  }

  return (
    <div className="max-w-md mx-auto flex flex-col gap-8 pt-6">
      <section>
        <p className="font-mono text-xs uppercase tracking-widest text-gray-500">
          Cuenta
        </p>
        <h1 className="font-display font-bold text-5xl tracking-tighter mt-4 mb-3">
          Crear cuenta<span className="text-accent">.</span>
        </h1>
        <p className="text-gray-700">Regístrate para comprar en TechStore.</p>
      </section>

      <form onSubmit={manejarEnvio} noValidate className="flex flex-col gap-5">
        <Campo etiqueta="Nombre completo" nombre="nombre" error={errores.nombre}>
          <input
            id="nombre"
            name="nombre"
            type="text"
            value={form.nombre}
            onChange={manejarCambio}
            placeholder="Tu nombre y apellido"
            autoComplete="name"
            className={claseInput(errores.nombre)}
          />
        </Campo>

        <Campo etiqueta="Correo electrónico" nombre="email" error={errores.email}>
          <input
            id="email"
            name="email"
            type="email"
            value={form.email}
            onChange={manejarCambio}
            placeholder="tucorreo@ejemplo.com"
            autoComplete="email"
            className={claseInput(errores.email)}
          />
        </Campo>

        <Campo etiqueta="Contraseña" nombre="password" error={errores.password}>
          <input
            id="password"
            name="password"
            type="password"
            value={form.password}
            onChange={manejarCambio}
            placeholder="Mínimo 6 caracteres"
            autoComplete="new-password"
            className={claseInput(errores.password)}
          />
        </Campo>

        <Campo etiqueta="Repite la contraseña" nombre="confirmar" error={errores.confirmar}>
          <input
            id="confirmar"
            name="confirmar"
            type="password"
            value={form.confirmar}
            onChange={manejarCambio}
            placeholder="Escríbela otra vez"
            autoComplete="new-password"
            className={claseInput(errores.confirmar)}
          />
        </Campo>

        <Campo etiqueta="Departamento" nombre="departamento" error={errores.departamento}>
          <select
            id="departamento"
            name="departamento"
            value={form.departamento}
            onChange={manejarCambio}
            className={claseInput(errores.departamento)}
          >
            <option value="">Selecciona un departamento</option>
            {departamentos.map((d) => (
              <option key={d.id} value={d.name}>
                {d.name}
              </option>
            ))}
          </select>
        </Campo>

        <Campo etiqueta="Municipio" nombre="municipio" error={errores.municipio}>
          <select
            id="municipio"
            name="municipio"
            value={form.municipio}
            onChange={manejarCambio}
            disabled={!form.departamento}
            className={claseInput(errores.municipio) + " disabled:opacity-50"}
          >
            <option value="">
              {form.departamento ? "Selecciona un municipio" : "Primero elige un departamento"}
            </option>
            {municipios.map((m) => (
              <option key={m.id} value={m.name}>
                {m.name}
              </option>
            ))}
          </select>
        </Campo>

        <button
          type="submit"
          disabled={estado === "enviando"}
          className="bg-ink text-paper px-7 py-3.5 rounded-card text-sm font-semibold enabled:hover:bg-accent transition-colors disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {estado === "enviando" ? "Creando cuenta..." : "Crear cuenta →"}
        </button>

        {estado === "error" && (
          <p role="alert" className="bg-gray-50 rounded-card p-4 text-sm font-semibold text-accent-dark">
            {mensajeError}
          </p>
        )}
      </form>

      <p className="text-sm text-gray-700">
        ¿Ya tienes cuenta?{" "}
        <Link to="/login" className="font-semibold underline hover:text-accent transition-colors">
          Ingresa
        </Link>
      </p>
    </div>
  );
}

export default Registro;

    if (form.nombre.trim().length < 3) {
      nuevos.nombre = "Escribe tu nombre completo (mínimo 3 caracteres)";
    }
    if (!/^\S+@\S+\.\S+$/.test(form.email)) {
      nuevos.email = "Escribe un correo válido, como nombre@gmail.com";
    } else {
      const dominio = form.email.slice(form.email.lastIndexOf("@") + 1).toLowerCase();
      if (!DOMINIOS_PERMITIDOS.includes(dominio)) {
        nuevos.email = "Usa un correo de Gmail, Hotmail, Outlook, Yahoo o iCloud";
      }
    }
    if (form.password.length < 6) {
      nuevos.password = "La contraseña debe tener al menos 6 caracteres";
    }
    if (form.confirmar !== form.password) {
      nuevos.confirmar = "Las contraseñas no coinciden";
    }
    if (!form.departamento) {
      nuevos.departamento = "Selecciona un departamento";
    }
    if (!form.municipio) {
      nuevos.municipio = "Selecciona un municipio";
    }
    return nuevos;
  }

  async function manejarEnvio(e) {
    e.preventDefault();

    const nuevos = validar();
    setErrores(nuevos);
    if (Object.keys(nuevos).length > 0) return;

    setEstado("enviando");
    try {
      const res = await fetch(URL_API + "/registro", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          nombre: form.nombre,
          email: form.email,
          password: form.password,
          departamento: form.departamento,
          municipio: form.municipio,
        }),
      });
      const datos = await res.json();
      if (!res.ok) throw new Error(datos.error || "No se pudo crear la cuenta");

      navegar("/login", {
        state: { aviso: "Cuenta creada. Ya puedes iniciar sesión." },
      });
    } catch (err) {
      setMensajeError(
        err instanceof TypeError ? "No se pudo conectar al servidor" : err.message,
      );
      setEstado("error");
    }
  }

  return (
    <div className="max-w-md mx-auto flex flex-col gap-8 pt-6">
      <section>
        <p className="font-mono text-xs uppercase tracking-widest text-gray-500">
          Cuenta
        </p>
        <h1 className="font-display font-bold text-5xl tracking-tighter mt-4 mb-3">
          Crear cuenta<span className="text-accent">.</span>
        </h1>
        <p className="text-gray-700">Regístrate para comprar en TechStore.</p>
      </section>

      <form onSubmit={manejarEnvio} noValidate className="flex flex-col gap-5">
        <Campo etiqueta="Nombre completo" nombre="nombre" error={errores.nombre}>
          <input
            id="nombre"
            name="nombre"
            type="text"
            value={form.nombre}
            onChange={manejarCambio}
            placeholder="Tu nombre y apellido"
            autoComplete="name"
            className={claseInput(errores.nombre)}
          />
        </Campo>

        <Campo etiqueta="Correo electrónico" nombre="email" error={errores.email}>
          <input
            id="email"
            name="email"
            type="email"
            value={form.email}
            onChange={manejarCambio}
            placeholder="tucorreo@ejemplo.com"
            autoComplete="email"
            className={claseInput(errores.email)}
          />
        </Campo>

        <Campo etiqueta="Contraseña" nombre="password" error={errores.password}>
          <input
            id="password"
            name="password"
            type="password"
            value={form.password}
            onChange={manejarCambio}
            placeholder="Mínimo 6 caracteres"
            autoComplete="new-password"
            className={claseInput(errores.password)}
          />
        </Campo>

        <Campo etiqueta="Repite la contraseña" nombre="confirmar" error={errores.confirmar}>
          <input
            id="confirmar"
            name="confirmar"
            type="password"
            value={form.confirmar}
            onChange={manejarCambio}
            placeholder="Escríbela otra vez"
            autoComplete="new-password"
            className={claseInput(errores.confirmar)}
          />
        </Campo>

        <Campo etiqueta="Departamento" nombre="departamento" error={errores.departamento}>
          <select
            id="departamento"
            name="departamento"
            value={form.departamento}
            onChange={manejarCambio}
            className={claseInput(errores.departamento)}
          >
            <option value="">Selecciona un departamento</option>
            {departamentos.map((d) => (
              <option key={d.id} value={d.name}>
                {d.name}
              </option>
            ))}
          </select>
        </Campo>

        <Campo etiqueta="Municipio" nombre="municipio" error={errores.municipio}>
          <select
            id="municipio"
            name="municipio"
            value={form.municipio}
            onChange={manejarCambio}
            disabled={!form.departamento}
            className={claseInput(errores.municipio) + " disabled:opacity-50"}
          >
            <option value="">
              {form.departamento ? "Selecciona un municipio" : "Primero elige un departamento"}
            </option>
            {municipios.map((m) => (
              <option key={m.id} value={m.name}>
                {m.name}
              </option>
            ))}
          </select>
        </Campo>

        <button
          type="submit"
          disabled={estado === "enviando"}
          className="bg-ink text-paper px-7 py-3.5 rounded-card text-sm font-semibold enabled:hover:bg-accent transition-colors disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {estado === "enviando" ? "Creando cuenta..." : "Crear cuenta →"}
        </button>

        {estado === "error" && (
          <p role="alert" className="bg-gray-50 rounded-card p-4 text-sm font-semibold text-accent-dark">
            {mensajeError}
          </p>
        )}
      </form>

      <p className="text-sm text-gray-700">
        ¿Ya tienes cuenta?{" "}
        <Link to="/login" className="font-semibold underline hover:text-accent transition-colors">
          Ingresa
        </Link>
      </p>
    </div>
  );
}
