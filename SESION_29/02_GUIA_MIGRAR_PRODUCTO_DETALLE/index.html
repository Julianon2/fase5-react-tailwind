¿Qué construimos hoy?
Una página propia para cada producto

Una sola página, ProductoDetalle.jsx, que muestra un producto distinto según la dirección: /productos/ seguido del _id. Aparecen tres ideas nuevas: la ruta con parámetro, el useEffect que depende de un dato y los estados de una página que carga.

Paso 1 · Ruta

Una ruta con parámetro y una página nueva

Hasta hoy cada ruta era una dirección fija. Una tienda no puede crear una ruta por producto: se escribe una sola con un hueco que cambia (:id) y una página que lee ese hueco. En este paso la página todavía no pide nada al backend: solo demuestra que lee la dirección.

1

Levantar los 2 servidores, cada uno en su propia terminal (si ya estaban corriendo, no hace falta reiniciarlos):

terminal 1 — dentro de backend/
npm run dev
terminal 2 — dentro de frontend-react/
npm run dev
Cómo verificar: localhost:5173/productos muestra las tarjetas, el buscador, el menú de orden y el modal funcionando, como quedó en S28.

Paso 2 · Datos

Pedir el producto al backend

La página ya sabe qué producto le piden. Ahora tiene que traerlo del backend. Es el useEffect + fetch de S24, con dos diferencias: pide un producto y se vuelve a ejecutar cuando el id cambia. Además, una página que carga datos tiene varias "caras": cargando, error y lista.

1

Antes de escribir código necesitas el id de un producto real. Abre en una pestaña nueva localhost:3000/api/productos y mira el primer producto de la lista:

respuesta del backend (ejemplo)
[
  {
    "_id": "6a726af1856112825e9c9f36",
    "id": 1,
    "nombre": "MacBook Pro M3",
    ...
  }
]
Cada producto tiene dos identificadores distintos. id (1, 2, 3...) es el número del catálogo: el que se ve en "REF 01" y el que usa el carrito. _id es el identificador que crea MongoDB, una cadena larga. El backend busca con findById, que usa el _id, así que la dirección de la página lleva el _id.

Copia el valor de _id de cualquier producto: lo usarás para probar este paso.

frontend-react/src/ProductoDetalle.jsxdespués — así queda el archivo
import { useState, useEffect } from "react";
import { useParams, Link } from "react-router-dom";

function ProductoDetalle() {
  const { id } = useParams();
  const [producto, setProducto] = useState(null);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    setCargando(true);
    setError(null);
    fetch("http://localhost:3000/api/productos/" + id)
      .then((res) => {
        if (!res.ok) throw new Error("No encontrado");
        return res.json();
      })
      .then((datos) => setProducto(datos))
      .catch(() => setError("Producto no encontrado"))
      .finally(() => setCargando(false));
  }, [id]);

  if (cargando) {
    return <p className="text-center py-20 text-gray-500">Cargando producto...</p>;
  }

  if (error) {
    return (
      <div className="flex flex-col items-center text-center gap-4 py-20">
        <p className="text-accent-dark">{error}</p>
        <Link
          to="/productos"
          className="font-semibold underline hover:text-accent transition-colors"
        >
          Volver al catálogo
        </Link>
      </div>
    );
  }

  return (
    <div>
      <h1 className="font-display font-bold text-3xl tracking-tight">
        {producto.nombre}
      </h1>
      <p className="text-gray-700 mt-2">{producto.descripcion}</p>
    </div>
  );
}

export default ProductoDetalle;

import { useParams, Link }Link navega a otra ruta sin recargar la página (S25). Se usa en la pantalla de error

const [cargando, setCargando] = useState(true)Empieza en true porque lo primero que hace la página es esperar la respuesta
const [error, setError] = useState(null)Guarda el mensaje si algo sale mal. null significa "sin error"

Los hooks (useState, useEffect, useParams) van siempre arriba, antes de cualquier return. Si pusieras un return antes de un hook, React perdería la cuenta de los hooks entre un dibujo y otro y fallaría.

frontend-react/src/ProductoDetalle.jsxdespués — así queda el archivo
import { useState, useEffect, useContext } from "react";
import { useParams, Link } from "react-router-dom";
import { CartContext } from "./CartContext.jsx";

function ProductoDetalle() {
  const { id } = useParams();
  const { agregarAlCarrito } = useContext(CartContext);
  const [producto, setProducto] = useState(null);
  const [cargando, setCargando] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    setCargando(true);
    setError(null);
    fetch("http://localhost:3000/api/productos/" + id)
      .then((res) => {
        if (!res.ok) throw new Error("No encontrado");
        return res.json();
      })
      .then((datos) => setProducto(datos))
      .catch(() => setError("Producto no encontrado"))
      .finally(() => setCargando(false));
  }, [id]);

  function agregar() {
    agregarAlCarrito({
      id: producto.id,
      nombre: producto.nombre,
      precio: producto.precio,
      imagen: producto.imagen,
    });
  }

  if (cargando) {
    return <p className="text-center py-20 text-gray-500">Cargando producto...</p>;
  }

  if (error) {
    return (
      <div className="flex flex-col items-center text-center gap-4 py-20">
        <p className="text-accent-dark">{error}</p>
        <Link
          to="/productos"
          className="font-semibold underline hover:text-accent transition-colors"
        >
          Volver al catálogo
        </Link>
      </div>
    );
  }

  return (
    <div className="flex flex-col gap-8">
      <Link
        to="/productos"
        className="text-sm font-semibold text-gray-700 hover:text-ink transition-colors"
      >
        ← Volver al catálogo
      </Link>

      <div className="grid gap-10 md:grid-cols-2">
        <img
          src={producto.imagen.replace("w=400&h=250", "w=1000&h=1250")}
          alt={producto.nombre}
          className="w-full aspect-4/5 object-cover rounded-3xl"
        />

        <div className="flex flex-col gap-5 md:py-6">
          <span className="font-mono text-[11px] font-bold tracking-wider text-gray-500">
            REF {String(producto.id).padStart(2, "0")}
          </span>
          <h1 className="font-display font-bold text-4xl md:text-5xl tracking-tighter leading-tight">
            {producto.nombre}
          </h1>
          <p className="text-gray-700 text-lg leading-relaxed">
            {producto.descripcion}
          </p>
          <p className="font-mono text-3xl font-bold">{producto.precio}</p>
          <button
            onClick={agregar}
            className="mt-4 bg-ink text-paper rounded-card py-4 text-sm font-semibold hover:bg-accent transition-colors"
          >
            Agregar al carrito
          </button>
        </div>
      </div>
    </div>
  );
}

export default ProductoDetalle;
